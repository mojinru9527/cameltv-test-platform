import { test, expect } from '@playwright/test';

test('进入直播间默认播放视频流', async ({ page }) => {
  await page.goto('http://localhost:5211/login');
  await page.waitForTimeout(1 * 1000);
  await expect(page.locator('body')).toContainText('CamelTv 测试平台');
  await page.fill('#username', 'admin');
  await page.fill('#password', '***REDACTED***');
  await page.getByText('登录').first().click();
  await page.waitForTimeout(3 * 1000);
  await expect(page.locator('body')).toContainText('工作台');
  await expect(page.locator('body')).toContainText('工作台');
  await page.screenshot({ path: 'playground-screenshot.png' });
});
